عبارت
 $lg^*(n)$
  به این معنی است که از n چند بار باید لگاریتم گرفت تا به عددی کمتر یا مساوی ۱ تبدیل شود.
}

			$f_2(n) = lg(n^{lg n}) $&
			$f_1(n) = \sqrt{lg n}$\\
			$f_6(n) = lg(n!) $&
			$f_5(n) = nlg(n)$&
			$f_4(n) = n^{\frac{1}{lg n}}$\\
			$f_9(n) = lg^{*}(n)$ &
			$f_8(n) = lg^{*}(lg(n))$&
			$f_7(n) = ln(n)$
		\end{tabular}
