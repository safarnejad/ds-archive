با استفاده از درخت بازگشت، پیچیدگی تابع $T(n)$ را در هر قسمت به دست آورید. نوشتن جواب نهایی کفایت می‌کند و نیاز به توضیح و اثبات نیست.
\شروع{زیرمسائل}
\زیرمسئله[4]
$T(n) = T(n/2) + n^2$

\پاسخ{
		$\Theta(n^2)$.
	اگر درخت بازگشت را رسم کنیم، در طبقۀ اول هزینۀ
	$n^2$
	در طبقۀ دوم هزینۀ 
	$n^2/4$
	و به همین ترتیب در طبقۀ $k$اُم هزینۀ 
	$n^2/k^2$
	داریم. اگر همۀ این هزینه‌ها را جمع کنیم، جواب یک ضریب ثابت از 
	$n^2$
	می شود.
	
}
\زیرمسئله[4]
$T(n) =3T(n/4) + 5n$

\پاسخ{
		$\Theta(n)$.
	تعداد طبقات در درخت بازگشت برابر 
	$\log_{4}n$
	است؛ پس در کل تعداد برگ‌هایمان برابر 
	$3^{\log_{4}n} = n^{\log_{4}3}$
	است که از $n$ کمتر است و جمع هزینه‌هایمان در کل طبقات هم برابر می‌شود با
	$5n(1+3/4+ 9/16 + ...)$.
	پس جواب 
	$\Theta(n)$
	می شود.
	
}

\زیرمسئله[4]
$T(n) = 2T(n-1) + \Theta(n)$

\پاسخ{
		$\Theta(2^{n})$.
	تعداد طبقات درخت بازگشت برابر $n$ است پس در کل 
	$2^n$
	تا برگ داریم. 
	جمع هزینه‌هایمان در کل طبقات هم برابر 
	$n + 2(n-1) + 4(n-2) + ... $
	است که برابر است با 
	$\Theta(2^n)$.
	پس در کل جواب هم 
	$\Theta(2^n)$
	است.
	
}
