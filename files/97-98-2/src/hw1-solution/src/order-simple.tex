$f_1(n)$

$f_4(n)$

$$f_2(n) < f_4(n) < f_1(n) < f_3(n) $$

\زیرمسئله[4]
\begin{large}
		\begin{center}
			\begin{tabular}{l l}
				$f_2(n) = n^5\cdot5^n $&
				$f_1(n) = n\sum\limits_{i=1}^n \frac{1}{\sqrt{i} + \sqrt{i+1}} $\\
				$f_4(n) = \sqrt{2^{\sqrt{n}}}$&
				$f_3(n) = 2^{\sqrt[n]{n^{n+1}}} $
			\end{tabular}
		\end{center}
\end{large}
\پاسخ{
		با توجه به این که قسمت نمایی 
	$f_2(n)$
	از بقیه بیشتر است، بنابراین از بقیه توابع رشد بیشتری دارد.
	از طرفی 
	$f_1(n)$
	از مرتبه 
	$n\sqrt{n}$
	می‌باشد. بنابراین از بقیه که نمایی‌اند کوچکتر است.
	همچنین به دلیل این که رشد توان 
	$f_3(n)$
	از 
	$f_4(n)$
	بیشتر است، بنابراین داریم:
	$$f_1(n) < f_4(n) < f_3(n) < f_2(n) $$
	
}
\زیرمسئله[4]

\begin{large}
		\begin{center}
			\begin{tabular}{l l}
				$f_2(n) = n\sqrt{n} $&
				$f_1(n) = 2^{\log_2 n^2} $\\
				$f_4(n) = \sqrt{n}\log n$&
				$f_3(n) = n\binom{n}{2} $
			\end{tabular}
		\end{center}
\end{large}
\پاسخ{
		می‌دانیم
	$2^{\log_2 n^2} = n^2$.
	همچنین 
	$f_3(n)$
	از مرتبه $n^3$ می‌باشد و رشد این تابع از بقیه بیشتر است.
	بنابراین داریم:
	$$f_4(n) < f_2(n) < f_1(n) < f_3(n) $$
	
}
