مثال:

‌دنباله‌ی داده شده
\begin{center}
\متن‌لاتین{10, 3, 4, 5, 1, 7, 6}
\end{center}

دنباله‌ی حاصل
\begin{center}
\متن‌لاتین{1, 4, 4, 4, 7, 6, 7}
\end{center}
