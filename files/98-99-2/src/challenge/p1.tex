
\شروع‌مسئله 
فرض کنید که
 \lr{q}
 درخواست وجود دارد. درخواست
\lr{i}
ام شامل 
${x_i, y_i}$
است که نشان‌دهنده‌ی این است یکی از کارمندان در این مختصات قرار دارد. شما باید پاسخ دهید در اتاقی که این کارمند در آن قرار دارد، چند در وجود دارد. 
دقت کنید که در نیمی از تست‌های هر بخش نیاز است که شما به شکل آنلاین(یعنی تا وقتی پاسخ یک درخواست را نداده اید نمی‌توانید درخواست بعدی را بیابید) و در نیمی دیگر به شکل آفلاین(یعنی درخواست ها از هم مستقل هستند) پاسخ دهید.

\متن‌سیاه{ورودی:}

در ابتدا یک
${n}$
 وارد می‌شود. در هر یک از
 ${n}$ 
خط بعدی، ۴ عدد صحیح می‌آیند که به ترتیب
 ${( x,y )}$
 نقطه‌ی شروع دیوار
 ${i}$
ام  و
 ${( x,y )}$
 پایان دیوار
 ${i}$
ام است. 
سپس 
${q}$ 
و بولین 
${type}$
 وارد می‌شود.  در هر یک
 ${q}$
 خط بعدی 
${x_i}$ 
و
 ${y_i}$
 وارد می‌شود.

\متن‌سیاه{خروجی:}

در صورتی که ${type}$ صفر بود، برای درخواست ${i}$ ام، مقدار ورودی همان مختصات کارمند است. (آفلاین)

درصورتی که ${type}$ برابر یک بود، برای درخواست ${i}$ ام، مقدار ورودی ها را با توان دوی خروجی درخواست قبلی ${xor}$ کنید. سپس مقدار های بدست آمده مختصات های کارمند در درخواست ${i}$ ام است.(آنلاین)

در هر دو حالت، در ${q}$ خط جواب‌های درخواست‌ها را خروجی دهید.

\متن‌سیاه{محدودیت‌های بخش اول ( ۱۶ نمره ): }

\begin{center}
    ${n,q \leq 3 \times 10^3 }$
\end{center}

\متن‌سیاه{محدودیت‌های بخش دوم ( ۲۴ نمره ): }


\begin{center}
    ${n,q \leq 3 \times 10^5 }$
\end{center}

تمامی مقادیر دیگر ورودی صحیح و در بازه‌ی $ [ -10 ^ {18} , 10 ^ {18} ] $قرار دارند.

\شروع‌مسئله 

فرض کنید که ${q}$ درخواست وجود دارد. هر درخواست، به شما ${(x1_i , y1_i)}$ و ${(x2_i, y2_i)}$ را می دهد که مختصات دو کارمند است. کارمند اول می‌خواهد که به کارمند دوم برسد. برای این کار می‌تواند در فضای اتاق‌ها حرکت کند یا با عبور از در، به سمت دیگر دیواری برود. فاصله‌ی کارمند دوم از کارمند اول را مینیمم تعداد درهایی که کارمند اول باید از آن بگذرد تا به کارمند دوم برسد، می‌نامیم. شما باید در مورد فاصله‌ی کارمند نفر دوم به سوالات زیر پاسخ دهید.
دقت کنید که در نیمی از تست‌های هر بخش نیاز است که شما به شکل آنلاین(یعنی تا وقتی پاسخ یک درخواست را نداده اید نمی‌توانید درخواست بعدی را بیابید) و در نیمی دیگر به شکل آفلاین(یعنی درخواست ها از هم مستقل هستند) پاسخ دهید.

\begin{itemize}
	\item \textbf{ مرحلۀ اول ( ۱۲ نمره ):}
	بگویید آیا فاصله‌ی کارمند دوم صفر است یا خیر ؟
	\item \textbf{ مرحلۀ دوم ( ۱۲ نمره ):}
	 بگویید آیا فاصله‌ی دو نفر حداکثر یک است یا خیر؟
	\item \textbf{ مرحلۀ سوم (۱۸ نمره ):}
	فاصله‌ی نفر دوم را مشخص کنید.  در این حالت تمامی دیوار ها عمودی و افقی هستند.
	\item \textbf{ مرحلۀ چهارم (۱۸ نمره ):}
	فاصله‌ی نفر دوم را مشخص کنید. این حالت محدودیت اضافی ندارد.
\end{itemize}


\متن‌سیاه{ورودی:}
در ابتدا یک ${n}$ وارد می‌شود. در هر یک از ${n}$  خط بعدی، ۴ عدد صحیح می‌آیند که به ترتیب ${(x_i , y_i)}$  نقطه‌ی شروع دیوار ${i}$ام  و ${(x1_i , y1_i)}$ پایان دیوار ${i}$ام است. 
سپس ${q}$ و بولین ${type}$ وارد می‌شود.  در هر یک ${q}$ خط بعدی ${(x1_i , y1_i)}$ و ${(x2_i, y2_i)}$ وارد می‌شود.


\متن‌سیاه{خروجی:}

\begin{itemize}
	\item \textbf{ مرحله‌ی اول و دوم :}
	
	در صورتی که ${type}$  صفر بود، برای درخواست${i}$ام ، مقدار ورودی همان مختصات کارمند است. (آفلاین)
	
درصورتی که ${type}$  برابر یک بود، برای درخواست ${i}$ام ، مقدار ورودی ها را با مختصات کارمند ${i-1}$ام (درصورت وجود و ${yes}$ بودن جواب درخواست ${i-1}$ام  ) ${xor}$ کنید. سپس مقدار های بدست آمده مختصات های کارمند در درخواست${i}$ام  است.(آنلاین)

در هر دو حالت، در ${q}$  خط جواب‌ها را  در به صورت ${yes}$ یا ${no}$ چاپ کنید.

	\item \textbf{ مرحله‌ی سوم و چهارم:}
	
	در صورتی که ${type}$  صفر بود، برای درخواست${i}$ام، مقدار ورودی همان مختصات کارمندها است. (آفلاین)

درصورتی که ${type}$ برابر یک بود، برای درخواست ${i}$ام، مقدار ورودی‌ها را با توان دوی خروجی درخواست قبلی ${xor}$ کنید. سپس مقدار های بدست آمده مختصات های کارمند‌ها در درخواست${i}$ام است.(آنلاین)

در هر دو حالت، در ${q}$ خط جواب‌های درخواست‌ها را خروجی دهید.

\end{itemize}


\متن‌سیاه{محدودیت‌ها: }


\begin{center}
    ${n,q \leq 5 \times 10^4 }$
\end{center}

تمامی مقادیر دیگر ورودی صحیح و در بازه‌ی $ [ -10 ^ {18} , 10 ^ {18} ] $قرار دارند.
